    mutable selected = pick;
    discovered.add(pick.code);
    mutable discovered = new Set(discovered);

  bar.querySelector("#reveal-btn").onclick = () => {

viewof region_filter = Inputs.radio(
  ["All Regions", "Americas", "Europe", "Asia", "Africa", "Oceania", "Antarctic"],
  { label: "🌍 Region:", value: "All Regions" }
)

viewof search_query = Inputs.text({
  label: "🔍 Search:",
  placeholder: "Search by country, capital, or language (reveals matching flags)..."
})
filtered_flags = flags.filter(d => {
  const matchesRegion = region_filter === "All Regions" || d.region === region_filter;
  const q = search_query.trim().toLowerCase();
  const matchesSearch = !q ||
    d.name.toLowerCase().includes(q) ||
    d.capital.toLowerCase().includes(q) ||
    d.languages.toLowerCase().includes(q) ||
    d.code.toLowerCase().includes(q);
  return matchesRegion && matchesSearch;
})

{
  const grid = html`<div style="
    display: grid;
    grid-template-columns: repeat(auto-fill, minmax(105px, 1fr));
    gap: 14px;
    margin-top: 1rem;
  "></div>`;

  for (const item of filtered_flags) {
    const isSelected = item.code === selected.code;
    const isRevealed = revealAll || isSelected || discovered.has(item.code) || search_query.trim().length > 0;
    const rc = regionColors[item.region] || { bg: "rgba(148,163,184,0.15)", border: "#64748b", text: "#cbd5e1" };

    const card = html`<button class="mini-flag-btn" title="Click to reveal country & fun facts!" style="
      background: ${isSelected
        ? "linear-gradient(145deg, rgba(37, 99, 235, 0.45), rgba(124, 58, 237, 0.45))"
        : isRevealed
        ? "rgba(30, 41, 59, 0.9)"
        : "rgba(15, 23, 42, 0.75)"};
      border: ${isSelected
        ? "2px solid #38bdf8"
        : isRevealed
        ? `1px solid ${rc.border}`
        : "1px solid rgba(148, 163, 184, 0.2)"};
      box-shadow: ${isSelected ? "0 0 18px rgba(56, 189, 248, 0.45)" : "0 2px 6px rgba(0,0,0,0.25)"};
    ">
      <img src="w2560/${item.file}" loading="lazy" style="
        width: 82px;
        height: 54px;
        object-fit: contain;
        border-radius: 4px;
        filter: drop-shadow(0 2px 4px rgba(0,0,0,0.4));
      ">
      <span style="
        margin-top: 8px;
        font-size: 0.76rem;
        font-weight: ${isRevealed ? "700" : "600"};
        color: ${isSelected ? "#ffffff" : isRevealed ? rc.text : "#64748b"};
        text-align: center;
        line-height: 1.15;
      ">${isRevealed ? item.name : "❓ Guess Me"}</span>
    </button>`;

    card.onclick = () => {
      mutable selected = item;
      discovered.add(item.code);
      mutable discovered = new Set(discovered);
      window.scrollTo({ top: 0, behavior: "smooth" });
    };
    grid.appendChild(card);
  }

  return grid;
}
